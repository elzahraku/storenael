⚙️ Konfigurasi

Jalankan SQL ini di Supabase → SQL Editor: -- 1. Tabel produk CREATE TABLE IF NOT EXISTS products ( id UUID DEFAULT gen_random_uuid() PRIMARY KEY, name TEXT NOT NULL, description TEXT, price INTEGER NOT NULL, image_url TEXT, category TEXT DEFAULT 'Umum', label_garansi TEXT DEFAULT 'Garansi 30 Hari', is_hot BOOLEAN DEFAULT false, created_at TIMESTAMPTZ DEFAULT NOW() ); -- 2. Aktifkan RLS ALTER TABLE products ENABLE ROW LEVEL SECURITY; -- 3. Policy tabel produk (baca, tambah, edit, hapus) DROP POLICY IF EXISTS "baca_produk" ON products; DROP POLICY IF EXISTS "tambah_produk" ON products; DROP POLICY IF EXISTS "update_produk" ON products; DROP POLICY IF EXISTS "hapus_produk" ON products; CREATE POLICY "baca_produk" ON products FOR SELECT USING (true); CREATE POLICY "tambah_produk" ON products FOR INSERT WITH CHECK (true); CREATE POLICY "update_produk" ON products FOR UPDATE USING (true); CREATE POLICY "hapus_produk" ON products FOR DELETE USING (true); -- 4. Storage bucket untuk foto produk INSERT INTO storage.buckets (id, name, public) VALUES ('products', 'products', true) ON CONFLICT DO NOTHING; -- 5. Policy storage (upload, lihat, update, hapus) DROP POLICY IF EXISTS "upload_foto" ON storage.objects; DROP POLICY IF EXISTS "lihat_foto" ON storage.objects; DROP POLICY IF EXISTS "update_foto" ON storage.objects; DROP POLICY IF EXISTS "hapus_foto" ON storage.objects; CREATE POLICY "upload_foto" ON storage.objects FOR INSERT WITH CHECK (bucket_id = 'products'); CREATE POLICY "lihat_foto" ON storage.objects FOR SELECT USING (bucket_id = 'products'); CREATE POLICY "update_foto" ON storage.objects FOR UPDATE USING (bucket_id = 'products'); CREATE POLICY "hapus_foto" ON storage.objects FOR DELETE USING (bucket_id = 'products');

Pelanggan akan diarahkan ke t.me/[username] saat checkout.
